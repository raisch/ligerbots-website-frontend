/** @module */
import createDebugMessages from 'debug'
import { getBackendClient } from '$lib/server/client'

import Queries from '$lib/server/graphql/event'
const debug = createDebugMessages('APP:lib/server/tripride')

/** @class */
export default class Vehicle {
  /**
   * Get all vehicles.
   *
   * @param {string} [query] - Custom GraphQL query to use.
   *
   * @returns {Promise<any[]>} - Array of vehicle records.
   *
   * @throws {Error} if failed to retrieve vehicles.
   */
  static async getAllVehicles(query = Queries.GET_ALL_VEHICLES_QUERY) {
    const client = await getBackendClient()

    if (!client) {
      throw new Error('Backend client is not available')
    }

    debug(`getAllVehicles() query: ${query}`)

    let result
    try {
      result = await client.query(query, {})
      debug(`getAllVehicles() resp: ${JSON.stringify(result)}`)
      result = result?.vehicle || []
    } catch (/** @type {any} */ err) {
      throw new Error(`Failed to retrieve vehicles: ${JSON.stringify(err)}`)
    }

    debug(`getAllVehicles() result: ${JSON.stringify(result)}`)
    return result
  }

  /**
   * Get a vehicle by ID.
   *
   * @param {string} vehicleId - The ID of the vehicle to retrieve.
   * @param {string} [query=Queries.GET_VEHICLE_BY_ID_QUERY] - The GraphQL query to use.
   *
   * @returns {Promise<Object>} - The vehicle record.
   *
   * @throws {Error} if failed to retrieve the vehicle.
   */
  static async getVehicleById(vehicleId, query = Queries.GET_VEHICLE_BY_ID_QUERY) {
    if (!vehicleId) {
      throw new Error('Vehicle ID is required')
    }

    const client = await getBackendClient()

    if (!client) {
      throw new Error('Backend client is not available')
    }

    const variables = {
      id: vehicleId
    }

    debug(`getVehicleById(vehicleId=${vehicleId}) query: ${query}`)
    debug(`getVehicleById(vehicleId=${vehicleId}) variables: ${JSON.stringify(variables)}`)

    let result
    try {
      result = await client.query(query, variables)
      debug(`getVehicleById(vehicleId=${vehicleId}) resp: ${JSON.stringify(result)}`)
      result = result?.vehicle_by_id || {}
    } catch (/** @type {any} */ err) {
      throw new Error(`Failed to retrieve vehicle: ${JSON.stringify(err)}`)
    }

    debug(`getVehicleById(vehicleId=${vehicleId}) result: ${JSON.stringify(result)}`)
    return result
  }
}
